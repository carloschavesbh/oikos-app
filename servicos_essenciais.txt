"""
====================================================================
MÃ“DULO DE SERVIÃ‡O DE SALDO (saldo_service.py)
====================================================================
ServiÃ§o centralizado para gestÃ£o de saldos bancÃ¡rios, marcos iniciais,
ajustes de conciliaÃ§Ã£o e transferÃªncias.

Garantias de Arquitetura:
- PrecisÃ£o monetÃ¡ria usando Decimal e quantizaÃ§Ã£o bancÃ¡ria (ROUND_HALF_EVEN).
- PersistÃªncia oficial de ajustes em centavos inteiros (INTEGER).
- IdentificaÃ§Ã£o relacional estÃ¡vel via conta_id (contas_config.id).
- CritÃ©rio determinÃ­stico de corte por snapshot para a data do marco.
- Isolamento contÃ¡bil: ajustes nÃ£o geram receitas/despesas familiares.
- Despesas com afeta_saldo_bancario=0 (desconto em folha) nÃ£o debitam contas bancÃ¡rias.
- Soft-delete para auditoria contÃ¡bil transparente.
"""

from decimal import Decimal, ROUND_HALF_EVEN

CENTAVOS = Decimal("0.01")


def to_decimal(val: Any) -> Decimal:
    """Converte qualquer valor numÃ©rico ou string para Decimal com 2 casas decimais."""
    if val is None or val == "":
        return Decimal("0.00")
    try:
        return Decimal(str(val)).quantize(CENTAVOS, rounding=ROUND_HALF_EVEN)
    except Exception:
        return Decimal("0.00")


def decimal_to_cents(d: Decimal) -> int:
    """Converte Decimal para centavos inteiros (ex: 150.25 -> 15025)."""
    return int((d * Decimal(100)).quantize(Decimal("1"), rounding=ROUND_HALF_EVEN))


def cents_to_decimal(cents: int) -> Decimal:
    """Converte centavos inteiros para Decimal com 2 casas decimais."""
    if cents is None:
        return Decimal("0.00")
    return (Decimal(cents) / Decimal(100)).quantize(CENTAVOS, rounding=ROUND_HALF_EVEN)

def obter_contas(conn: sqlite3.Connection, incluir_cofrinhos: bool = True) -> List[Dict[str, Any]]:
    """Retorna contas configuradas ordenadas por id."""
    cur = conn.cursor()
    cur.execute("PRAGMA table_info(contas_config)")
    cols = [r[1] for r in cur.fetchall()]
    tem_extras = "conta_pai_id" in cols and "incluir_saldo_consolidado" in cols
    if tem_extras:
        cur.execute("SELECT id, nome, tipo, cor, conta_pai_id, incluir_saldo_consolidado FROM contas_config ORDER BY id ASC")
        rows = cur.fetchall()
        res = []
        for r in rows:
            if not incluir_cofrinhos and r[2] == 'cofrinho':
                continue
            res.append({
                "id": r[0],
                "nome": r[1],
                "tipo": r[2],
                "cor": r[3],
                "conta_pai_id": r[4],
                "incluir_saldo_consolidado": r[5] if r[5] is not None else 1
            })
        return res
    else:
        cur.execute("SELECT id, nome, tipo, cor FROM contas_config ORDER BY id ASC")
        rows = cur.fetchall()
        return [{"id": r[0], "nome": r[1], "tipo": r[2], "cor": r[3], "conta_pai_id": None, "incluir_saldo_consolidado": 1} for r in rows]



def obter_ajuste_ativo(
    conn: sqlite3.Connection, 
    conta_id: int, 
    data_limite: Optional[str] = None
) -> Optional[Dict[str, Any]]:
    """
    Retorna o ajuste ativo mais recente para a conta (ordenado por data DESC, id DESC).
    Se data_limite for informada, considera apenas ajustes com data <= data_limite.
    """
    cur = conn.cursor()
    if data_limite:
        cur.execute("""
            SELECT id, conta_id, data, saldo_informado_centavos, saldo_calculado_centavos,
                   diferenca_centavos, tipo, transacoes_contempladas_ids, motivo, criado_em
            FROM ajustes_saldo
            WHERE conta_id = ? AND status = 'ativo' AND data <= ?
            ORDER BY data DESC, id DESC
            LIMIT 1
        """, (conta_id, data_limite))
    else:
        cur.execute("""
            SELECT id, conta_id, data, saldo_informado_centavos, saldo_calculado_centavos,
                   diferenca_centavos, tipo, transacoes_contempladas_ids, motivo, criado_em
            FROM ajustes_saldo
            WHERE conta_id = ? AND status = 'ativo'
            ORDER BY data DESC, id DESC
            LIMIT 1
        """, (conta_id,))
    
    row = cur.fetchone()
    if not row:
        return None

    contempladas = []
    if row[7]:
        try:
            contempladas = json.loads(row[7])
        except Exception:
            contempladas = []

    return {
        "id": row[0],
        "conta_id": row[1],
        "data": row[2],
        "saldo_informado": cents_to_decimal(row[3]),
        "saldo_informado_centavos": row[3],
        "saldo_calculado": cents_to_decimal(row[4]),
        "diferenca": cents_to_decimal(row[5]),
        "tipo": row[6],
        "transacoes_contempladas_ids": set(contempladas),
        "motivo": row[8],
        "criado_em": row[9]
    }


def calcular_saldo_conta(
    conn: sqlite3.Connection, 
    conta_id: int, 
    data_referencia: Optional[str] = None
) -> Dict[str, Any]:
    """
    Calcula o saldo real de uma conta com base no Ãºltimo marco/ajuste ativo e 
    nas movimentaÃ§Ãµes efetivadas posteriores.
    
    Regras de NegÃ³cio:
    1. Se houver marco/ajuste ativo:
       - Saldo inicial = saldo informado no marco.
       - LanÃ§amentos com data < data_marco: ignorados (jÃ¡ contemplados).
       - LanÃ§amentos com data == data_marco: considerados apenas se seu ID NÃƒO estiver
         em transacoes_contempladas_ids (ou seja, se for posterior ao corte).
       - LanÃ§amentos com data > data_marco (e <= data_referencia): somados/subtraÃ­dos.
    2. Se NÃƒO houver marco/ajuste:
       - Saldo inicial = 0.
       - Todas as movimentaÃ§Ãµes efetivadas (<= data_referencia) sÃ£o somadas/subtraÃ­das.
    3. MovimentaÃ§Ãµes com afeta_saldo_bancario = 0 (desconto em folha) NÃƒO alteram o saldo.
    4. TransferÃªncias debitam a conta de origem e creditam a conta de destino.
    """
    cur = conn.cursor()

    # InformaÃ§Ãµes da conta
    cur.execute("SELECT id, nome, tipo, cor FROM contas_config WHERE id = ?", (conta_id,))
    conta_row = cur.fetchone()
    if not conta_row:
        return {
            "conta_id": conta_id,
            "conta_nome": "Desconhecida",
            "saldo_atual": Decimal("0.00"),
            "saldo_centavos": 0,
            "tem_marco": False,
            "ajuste_id": None,
            "total_receitas_posteriores": Decimal("0.00"),
            "total_despesas_posteriores": Decimal("0.00"),
            "total_transf_entradas": Decimal("0.00"),
            "total_transf_saidas": Decimal("0.00")
        }
    
    conta_nome = conta_row[1]

    # Busca o marco/ajuste ativo mais recente
    marco = obter_ajuste_ativo(conn, conta_id, data_limite=data_referencia)

    if marco:
        saldo_cents = marco["saldo_informado_centavos"]
        data_corte = marco["data"]
        contempladas_ids = marco["transacoes_contempladas_ids"]
        tem_marco = True
        ajuste_id = marco["id"]
        marco_data = marco["data"]
        marco_saldo = marco["saldo_informado"]
    else:
        saldo_cents = 0
        data_corte = None
        contempladas_ids = set()
        tem_marco = False
        ajuste_id = None
        marco_data = None
        marco_saldo = None

    # Busca todas as movimentaÃ§Ãµes efetivadas que envolvem esta conta como origem ou destino
    # Respeita o campo afeta_saldo_bancario = 1 e pago = 1
    query = """
        SELECT id, data, valor, tipo, conta_id, conta_destino_id, afeta_saldo_bancario, conta
        FROM transacoes
        WHERE pago = 1 
          AND (afeta_saldo_bancario = 1 OR afeta_saldo_bancario IS NULL)
          AND (
              conta_id = ? 
              OR conta_destino_id = ?
              OR (conta_id IS NULL AND conta = ?)
          )
    """
    params = [conta_id, conta_id, conta_nome]

    if data_referencia:
        query += " AND data <= ?"
        params.append(data_referencia)

    cur.execute(query, params)
    rows = cur.fetchall()

    rec_post_cents = 0
    desp_post_cents = 0
    transf_in_cents = 0
    transf_out_cents = 0

    # Identifica despesas que possuem pagamentos registrados na tabela dedicada
    cur.execute("SELECT name FROM sqlite_master WHERE type='table' AND name='pagamentos_despesas'")
    tem_tabela_pagamentos = cur.fetchone() is not None

    ids_com_pagamentos = set()
    if tem_tabela_pagamentos:
        cur.execute("SELECT DISTINCT transacao_id FROM pagamentos_despesas WHERE cancelado = 0")
        ids_com_pagamentos = set(row[0] for row in cur.fetchall())

    for r in rows:
        t_id, t_data, t_val, t_tipo, t_origem_id, t_dest_id, t_afeta, t_conta_str = r
        val_cents = decimal_to_cents(to_decimal(t_val))

        # Se for despesa com pagamentos registrados em pagamentos_despesas, o dÃ©bito contÃ¡bil
        # virÃ¡ exclusivamente dos pagamentos efetivos para evitar dupla contagem.
        if t_tipo == "despesa" and t_id in ids_com_pagamentos:
            continue

        # Determina se a transaÃ§Ã£o Ã© posterior ao corte
        eh_posterior = False
        if not tem_marco:
            eh_posterior = True
        else:
            if t_data > data_corte:
                eh_posterior = True
            elif t_data == data_corte:
                # Na mesma data: sÃ³ afeta se NÃƒO constar na lista de contempladas
                if t_id not in contempladas_ids:
                    eh_posterior = True
            else:
                eh_posterior = False

        if not eh_posterior:
            continue

        # AplicaÃ§Ã£o das regras contÃ¡beis para transaÃ§Ãµes legadas/diretas
        eh_origem = (t_origem_id == conta_id) or (t_origem_id is None and t_conta_str == conta_nome)
        eh_destino = (t_dest_id == conta_id)

        if t_tipo == "receita" and eh_origem:
            saldo_cents += val_cents
            rec_post_cents += val_cents
        elif t_tipo == "despesa" and eh_origem:
            saldo_cents -= val_cents
            desp_post_cents += val_cents
        elif t_tipo == "transferencia":
            if eh_origem:
                saldo_cents -= val_cents
                transf_out_cents += val_cents
            if eh_destino:
                saldo_cents += val_cents
                transf_in_cents += val_cents

    # Incorpora pagamentos reais de despesas da tabela dedicada pagamentos_despesas
    if tem_tabela_pagamentos:
        q_pag = """
            SELECT id, transacao_id, data_pagamento, valor_pago_centavos, considerado_no_marco, afeta_saldo_bancario
            FROM pagamentos_despesas
            WHERE conta_id = ? 
              AND cancelado = 0 
              AND (afeta_saldo_bancario = 1 OR afeta_saldo_bancario IS NULL)
        """
        params_pag = [conta_id]
        if data_referencia:
            q_pag += " AND (data_pagamento IS NULL OR data_pagamento <= ?)"
            params_pag.append(data_referencia)
        cur.execute(q_pag, params_pag)
        pags = cur.fetchall()

        for p in pags:
            p_id, p_t_id, p_data, p_val_cents, p_considerado, p_afeta = p
            # Se o pagamento foi marcado como jÃ¡ refletido no marco inicial ou se a data real for desconhecida prÃ©-marco, nÃ£o desconta
            if p_considerado == 1 or p_data is None:
                continue

            eh_post_pag = False
            if not tem_marco:
                eh_post_pag = True
            else:
                if p_data > data_corte:
                    eh_post_pag = True
                elif p_data == data_corte:
                    # Na data do marco: se considerado_no_marco == 0, ocorreu apÃ³s a conciliaÃ§Ã£o
                    eh_post_pag = True
                else:
                    eh_post_pag = False

            if eh_post_pag:
                saldo_cents -= p_val_cents
                desp_post_cents += p_val_cents

    return {
        "conta_id": conta_id,
        "conta_nome": conta_nome,
        "saldo_atual": cents_to_decimal(saldo_cents),
        "saldo_centavos": saldo_cents,
        "tem_marco": tem_marco,
        "ajuste_id": ajuste_id,
        "marco_data": marco_data,
        "marco_saldo": marco_saldo,
        "total_receitas_posteriores": cents_to_decimal(rec_post_cents),
        "total_despesas_posteriores": cents_to_decimal(desp_post_cents),
        "total_transf_entradas": cents_to_decimal(transf_in_cents),
        "total_transf_saidas": cents_to_decimal(transf_out_cents),
    }


def calcular_saldo_consolidado(
    conn: sqlite3.Connection, 
    data_referencia: Optional[str] = None
) -> Dict[str, Any]:
    """
    Calcula o saldo consolidado de todas as contas ativas em contas_config.
    Separa rigorosamente o saldo livre disponÃ­vel do saldo alocado em reservas/cofrinhos,
    permitindo ao usuÃ¡rio configurar se cada cofrinho Ã© incluÃ­do no consolidado livre ou apenas nas reservas.
    TransferÃªncias internas anulam-se no total geral com reservas.
    """
    contas = obter_contas(conn, incluir_cofrinhos=True)
    total_livre_cents = 0
    total_reservas_cents = 0
    detalhes_contas = []

    for c in contas:
        res = calcular_saldo_conta(conn, c["id"], data_referencia=data_referencia)
        res["tipo"] = c.get("tipo", "corrente")
        res["conta_pai_id"] = c.get("conta_pai_id")
        res["incluir_saldo_consolidado"] = c.get("incluir_saldo_consolidado", 1)
        detalhes_contas.append(res)

        eh_cofrinho = (c.get("tipo") == "cofrinho")
        incluir_no_livre = (c.get("incluir_saldo_consolidado", 1 if not eh_cofrinho else 0) == 1)

        if not eh_cofrinho or incluir_no_livre:
            total_livre_cents += res["saldo_centavos"]
        else:
            total_reservas_cents += res["saldo_centavos"]

    total_geral_cents = total_livre_cents + total_reservas_cents

    return {
        "saldo_consolidado": cents_to_decimal(total_livre_cents),
        "saldo_consolidado_centavos": total_livre_cents,
        "saldo_reservas": cents_to_decimal(total_reservas_cents),
        "saldo_reservas_centavos": total_reservas_cents,
        "saldo_total_com_reservas": cents_to_decimal(total_geral_cents),
        "saldo_total_com_reservas_centavos": total_geral_cents,
        "contas": detalhes_contas
    }


def transferir_para_cofrinho(
    conn: sqlite3.Connection,
    cofrinho_id: int,
    valor: float,
    data_iso: str,
    descricao: Optional[str] = None
) -> Tuple[bool, str]:
    """
    Realiza transferÃªncia interna de recurso livre do Mercado Pago para um cofrinho vinculado.
    Reduz o saldo livre da origem e aumenta o saldo do cofrinho sem gerar receita ou despesa.
    """
    if valor <= 0:
        return False, "O valor deve ser maior que zero."

    cur = conn.cursor()
    cur.execute("SELECT id, nome, conta_pai_id, conta_config_id FROM cofrinhos WHERE id = ?", (cofrinho_id,))
    row = cur.fetchone()
    if not row:
        return False, "Cofrinho nÃ£o encontrado."

    cof_id, cof_nome, pai_id, cc_id = row
    origem_id = pai_id or 1
    destino_id = cc_id

    if not destino_id:
        cur.execute("SELECT id FROM contas_config WHERE nome = ?", (cof_nome,))
        row_cc = cur.fetchone()
        if row_cc:
            destino_id = row_cc[0]
            cur.execute("UPDATE cofrinhos SET conta_config_id = ? WHERE id = ?", (destino_id, cof_id))
        else:
            cur.execute("""
                INSERT INTO contas_config (nome, tipo, cor, conta_pai_id, incluir_saldo_consolidado)
                VALUES (?, 'cofrinho', '#f59e0b', ?, 0)
            """, (cof_nome, origem_id))
            destino_id = cur.lastrowid
            cur.execute("UPDATE cofrinhos SET conta_config_id = ? WHERE id = ?", (destino_id, cof_id))

    desc_final = descricao.strip() if descricao and descricao.strip() else f"Aporte {cof_nome}"

    cur.execute("""
        INSERT INTO transacoes (data, descricao, valor, tipo, categoria, conta, conta_id, conta_destino_id, pago, afeta_saldo_bancario)
        VALUES (?, ?, ?, 'transferencia', 'TransferÃªncia Interna', 'Mercado Pago', ?, ?, 1, 1)
    """, (data_iso, desc_final, float(valor), origem_id, destino_id))

    # Sincroniza saldo_confirmado em cofrinhos
    res_dest = calcular_saldo_conta(conn, destino_id)
    cur.execute("UPDATE cofrinhos SET saldo_confirmado = ?, data_atualizacao = ? WHERE id = ?", 
                (float(res_dest["saldo_atual"]), data_iso, cof_id))

    conn.commit()
    return True, f"R$ {valor:,.2f} guardado no {cof_nome} com sucesso!"


def retirar_de_cofrinho(

    """

    res_orig_pos = calcular_saldo_conta(conn, origem_id)

def registrar_ajuste(
    conn: sqlite3.Connection,
    conta_id: int,
    data_ajuste: str,
    saldo_informado: Decimal,
    tipo: str = "ajuste",
    motivo: str = "",
    transacoes_contempladas_ids: Optional[List[int]] = None
) -> int:
    """
    Registra um marco ou ajuste de saldo.
    Calcula o saldo que o sistema calculava atÃ© aquele momento exato para apurar a diferenÃ§a.
    Valores sÃ£o estritamente persistidos em centavos inteiros.
    """
    saldo_informado_dec = to_decimal(saldo_informado)
    saldo_informado_cents = decimal_to_cents(saldo_informado_dec)

    # Calcula saldo do sistema na data de referÃªncia
    res_atual = calcular_saldo_conta(conn, conta_id, data_referencia=data_ajuste)
    saldo_calculado_cents = res_atual["saldo_centavos"]
    diferenca_cents = saldo_informado_cents - saldo_calculado_cents

    ids_contemplados = transacoes_contempladas_ids or []
    json_contempladas = json.dumps(ids_contemplados)

    cur = conn.cursor()
    cur.execute("""
        INSERT INTO ajustes_saldo (
            conta_id, data, saldo_informado_centavos, saldo_calculado_centavos,
            diferenca_centavos, tipo, transacoes_contempladas_ids, status, motivo
        ) VALUES (?, ?, ?, ?, ?, ?, ?, 'ativo', ?)
    """, (
        conta_id,
        data_ajuste,
        saldo_informado_cents,
        saldo_calculado_cents,
        diferenca_cents,
        tipo,
        json_contempladas,
        motivo
    ))

    novo_id = cur.lastrowid
    conn.commit()
    return novo_id

def registrar_pagamento(
    conn: sqlite3.Connection,
    transacao_id: int,
    data_pagamento: Optional[str],
    valor_pago: float,
    conta_id: int,
    afeta_saldo_bancario: int = 1,
    considerado_no_marco: int = 0,
    forma_pagamento: str = "Pix",
    observacao: str = "",
    data_real_desconhecida: int = 0,
    classificacao: str = "pagamento_efetivo",
    tipo_ajuste_valor: str = "normal",
    valor_encargos: float = 0.0,
    valor_desconto: float = 0.0
) -> Dict[str, Any]:
    """
    Registra um pagamento (integral ou parcial) para uma obrigaÃ§Ã£o em transacoes.
    Preserva a data de vencimento da obrigaÃ§Ã£o original e vincula o dÃ©bito bancÃ¡rio
    Ã  conta utilizada e Ã  data efetiva do pagamento.
    
    Aplica validaÃ§Ã£o estrita (CondiÃ§Ã£o 13): impede que pagamentos ultrapassem o saldo
    devedor restante sem que haja declaraÃ§Ã£o explÃ­cita de encargos/juros ou renegociaÃ§Ã£o.
    """
    cur = conn.cursor()
    cur.execute("SELECT id, valor, data, descricao FROM transacoes WHERE id = ?", (transacao_id,))
    t = cur.fetchone()
    if not t:
        raise ValueError(f"ObrigaÃ§Ã£o #{transacao_id} nÃ£o encontrada.")

    val_orig = to_decimal(t[1])
    val_pago_dec = to_decimal(valor_pago)
    val_cents = decimal_to_cents(val_pago_dec)

    # Verifica saldo devedor atual antes do novo pagamento
    cur.execute("""
        SELECT COALESCE(SUM(valor_pago_centavos), 0)
        FROM pagamentos_despesas
        WHERE transacao_id = ? AND cancelado = 0
    """, (transacao_id,))
    total_pago_anterior_cents = cur.fetchone()[0]
    total_pago_anterior = cents_to_decimal(total_pago_anterior_cents)
    saldo_devedor_atual = max(Decimal("0.00"), val_orig - total_pago_anterior)

    # CondiÃ§Ã£o 13: ValidaÃ§Ã£o de limite do saldo devedor
    if tipo_ajuste_valor == "normal" and val_pago_dec > saldo_devedor_atual:
        excesso = val_pago_dec - saldo_devedor_atual
        raise ValueError(
            f"O valor do pagamento (R$ {val_pago_dec:.2f}) excede o saldo devedor restante da obrigaÃ§Ã£o (R$ {saldo_devedor_atual:.2f}) em R$ {excesso:.2f}. "
            f"Para efetuar pagamento maior que o saldo devedor, selecione explicitamente o tipo de ajuste (ex: 'Com Juros/Multa' ou 'RenegociaÃ§Ã£o/Acordo') e informe os encargos."
        )

    # 1. Insere o pagamento auditÃ¡vel
    cur.execute("""
        INSERT INTO pagamentos_despesas 
        (transacao_id, data_pagamento, data_real_desconhecida, classificacao, valor_pago, valor_pago_centavos, conta_id, afeta_saldo_bancario, considerado_no_marco, forma_pagamento, observacao, tipo_ajuste_valor, valor_encargos, valor_desconto)
        VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)
    """, (transacao_id, data_pagamento, data_real_desconhecida, classificacao, float(val_pago_dec), val_cents, conta_id, afeta_saldo_bancario, considerado_no_marco, forma_pagamento, observacao, tipo_ajuste_valor, float(valor_encargos), float(valor_desconto)))
    pag_id = cur.lastrowid

    # 2. Recalcula o total pago e atualiza status da obrigaÃ§Ã£o
    cur.execute("""
        SELECT COALESCE(SUM(valor_pago_centavos), 0)
        FROM pagamentos_despesas
        WHERE transacao_id = ? AND cancelado = 0
    """, (transacao_id,))
    total_pago_cents = cur.fetchone()[0]
    total_pago = cents_to_decimal(total_pago_cents)
    saldo_devedor = max(Decimal("0.00"), val_orig - total_pago)

    if total_pago >= val_orig:
        novo_status = "pago"
        novo_pago = 1
    elif total_pago > Decimal("0.00"):
        novo_status = "parcial"
        novo_pago = 2  # 2 indica parcialmente pago
    else:
        novo_status = "pendente"
        novo_pago = 0

    cur.execute("UPDATE transacoes SET pago = ?, status_pagamento = ? WHERE id = ?", (novo_pago, novo_status, transacao_id))
    conn.commit()

    return {
        "pagamento_id": pag_id,
        "transacao_id": transacao_id,
        "data_pagamento": data_pagamento,
        "valor_pago": val_pago_dec,
        "total_pago": total_pago,
        "saldo_devedor": saldo_devedor,
        "status_pagamento": novo_status,
        "pago_flag": novo_pago
    }


def estornar_pagamento(
    conn: sqlite3.Connection,
    pagamento_id: int,
    motivo: str = "Estorno solicitado pelo usuÃ¡rio"
) -> Dict[str, Any]:
    """
    Estorna um pagamento especÃ­fico com rastreabilidade (soft-delete).
    RecompÃµe o saldo bancÃ¡rio se foi pÃ³s-marco e atualiza o saldo devedor da obrigaÃ§Ã£o.
    """
    cur = conn.cursor()
    cur.execute("SELECT transacao_id, valor_pago, conta_id, data_pagamento, considerado_no_marco FROM pagamentos_despesas WHERE id = ?", (pagamento_id,))
    p = cur.fetchone()
    if not p:
        raise ValueError(f"Pagamento #{pagamento_id} nÃ£o encontrado.")

    t_id = p[0]

    # Marca como cancelado
    cur.execute("""
        UPDATE pagamentos_despesas
        SET cancelado = 1, cancelado_em = CURRENT_TIMESTAMP, motivo_cancelamento = ?
        WHERE id = ?
    """, (motivo, pagamento_id))

    # Recalcula a obrigaÃ§Ã£o vinculada
    cur.execute("SELECT valor FROM transacoes WHERE id = ?", (t_id,))
    val_orig = to_decimal(cur.fetchone()[0])

    cur.execute("""
        SELECT COALESCE(SUM(valor_pago_centavos), 0)
        FROM pagamentos_despesas
        WHERE transacao_id = ? AND cancelado = 0
    """, (t_id,))
    total_pago_cents = cur.fetchone()[0]
    total_pago = cents_to_decimal(total_pago_cents)
    saldo_devedor = max(Decimal("0.00"), val_orig - total_pago)

    if total_pago >= val_orig:
        novo_status = "pago"
        novo_pago = 1
    elif total_pago > Decimal("0.00"):
        novo_status = "parcial"
        novo_pago = 2
    else:
        novo_status = "pendente"
        novo_pago = 0

    cur.execute("UPDATE transacoes SET pago = ?, status_pagamento = ? WHERE id = ?", (novo_pago, novo_status, t_id))
    conn.commit()

    return {
        "transacao_id": t_id,
        "total_pago": total_pago,
        "saldo_devedor": saldo_devedor,
        "status_pagamento": novo_status,
        "pago_flag": novo_pago
    }

def obter_resumo_obrigacao(conn: sqlite3.Connection, transacao_id: int) -> Dict[str, Any]:
    """Retorna resumo completo da obrigaÃ§Ã£o com valor nominal, total pago, saldo devedor e lista de pagamentos."""
    cur = conn.cursor()
    cur.execute("SELECT id, data, descricao, valor, pago, status_pagamento, conta, conta_id FROM transacoes WHERE id = ?", (transacao_id,))
    t = cur.fetchone()
    if not t:
        return {}
    val_orig = to_decimal(t[3])
    pags = obter_pagamentos_transacao(conn, transacao_id)
    pags_ativos = [p for p in pags if not p["cancelado"]]
    total_pago = sum((p["valor_pago"] for p in pags_ativos), Decimal("0.00"))
    saldo_devedor = max(Decimal("0.00"), val_orig - total_pago)
    return {
        "id": t[0],
        "data_vencimento": t[1],
        "descricao": t[2],
        "valor_original": val_orig,
        "pago_flag": t[4],
        "status_pagamento": t[5] or ("pago" if t[4] == 1 else "pendente"),
        "conta_padrao": t[6],
        "conta_id": t[7],
        "total_pago": total_pago,
        "saldo_devedor": saldo_devedor,
        "pagamentos": pags,
        "pagamentos_ativos": pags_ativos
    }


def obter_metricas_combustivel(conn: sqlite3.Connection, data_referencia: Optional[str] = None) -> Dict[str, Any]:
    """
    Retorna mÃ©tricas segregadas de combustÃ­vel e abastecimentos do veÃ­culo (estilo Drivvo):
    1. Total efetivamente pago em abastecimentos (fluxo de caixa real) no dia, semana e mÃªs.
    2. Consumo operacional estimado por quilometragem rodada nos turnos.
    3. Faturamento bruto das corridas.
    4. Resultado de Caixa apÃ³s Abastecimentos (Faturamento - Abastecimento Pago).
    5. Resultado Operacional Estimado (Faturamento - Consumo Estimado).
    NÃƒO desconta os dois custos simultaneamente e nÃ£o os chama de lucro lÃ­quido completo.
    """
    if data_referencia:
        hoje = datetime.strptime(data_referencia, "%Y-%m-%d").date()
    else:
        hoje = date.today()
    hoje_iso = hoje.strftime("%Y-%m-%d")
    inicio_semana = (hoje - timedelta(days=hoje.weekday())).strftime("%Y-%m-%d")
    fim_semana = (hoje + timedelta(days=6 - hoje.weekday())).strftime("%Y-%m-%d")
    inicio_mes = hoje.replace(day=1).strftime("%Y-%m-%d")
    ultimo_dia_mes = calendar.monthrange(hoje.year, hoje.month)[1]
    fim_mes = hoje.replace(day=ultimo_dia_mes).strftime("%Y-%m-%d")

    cur = conn.cursor()
    
    # 1. Abastecimentos registrados em veiculo_abastecimentos
    cur.execute("SELECT name FROM sqlite_master WHERE type='table' AND name='veiculo_abastecimentos'")
    tem_tabela = cur.fetchone() is not None

    abastecimentos = []
    ids_transacoes_vinculadas = set()

    pago_hoje = 0.0
    pago_semana = 0.0
    pago_mes = 0.0

    litros_hoje = 0.0
    litros_semana = 0.0
    litros_mes = 0.0

    if tem_tabela:
        cur.execute("""
            SELECT id, data, hodometro, combustivel, litros, preco_litro, valor_bruto,
                   desconto, valor_pago, conta, conta_id, tanque_cheio, posto, observacao, transacao_id
            FROM veiculo_abastecimentos
            ORDER BY data DESC, hodometro DESC, id DESC
        """)
        for r in cur.fetchall():
            val_pago = float(r[8] or 0.0)
            lit = float(r[4]) if r[4] is not None else None
            hod = float(r[2]) if r[2] is not None else None
            prc = float(r[5]) if r[5] is not None else ((val_pago / lit) if lit and lit > 0 else None)
            tq = int(r[11]) if r[11] is not None else None
            dt = r[1]
            if r[14]:
                ids_transacoes_vinculadas.add(r[14])
            abastecimentos.append({
                "id": r[0],
                "data": dt,
                "hodometro": hod,
                "combustivel": r[3] or "Etanol",
                "litros": lit,
                "preco_litro": prc,
                "valor_bruto": float(r[6] or val_pago),
                "desconto": float(r[7] or 0.0),
                "valor_pago": val_pago,
                "conta": r[9] or "Conta",
                "conta_id": r[10],
                "tanque_cheio": tq,
                "posto": r[12] or "",
                "observacao": r[13] or "",
                "transacao_id": r[14]
            })
            if dt == hoje_iso:
                pago_hoje += val_pago
                if lit is not None:
                    litros_hoje += lit
            if inicio_semana <= dt <= fim_semana:
                pago_semana += val_pago
                if lit is not None:
                    litros_semana += lit
            if inicio_mes <= dt <= fim_mes:
                pago_mes += val_pago
                if lit is not None:
                    litros_mes += lit

    # TambÃ©m verifica se hÃ¡ despesas de combustÃ­vel em transacoes ainda nÃ£o cadastradas em veiculo_abastecimentos
    cur.execute("""
        SELECT id, data, descricao, valor, conta, conta_id, tipo, categoria
        FROM transacoes
        WHERE tipo = 'despesa'
          AND pago = 1
          AND (
              categoria LIKE '%Combust%'
              OR descricao LIKE '%Combust%'
              OR descricao LIKE '%Abastec%'
              OR descricao LIKE '%Etanol%'
              OR descricao LIKE '%Gasolina%'
          )
        ORDER BY data DESC, id DESC
    """)
    despesas_comb_avulsas = []
    for tr in cur.fetchall():
        tr_id, dt, desc, val, c_nome, c_id, tp, cat = tr
        if tr_id not in ids_transacoes_vinculadas:
            val_f = float(val or 0.0)
            despesas_comb_avulsas.append({
                "id": tr_id,
                "data": dt,
                "descricao": desc,
                "valor": val_f,
                "conta": c_nome,
                "conta_id": c_id,
                "categoria": cat
            })
            if dt == hoje_iso:
                pago_hoje += val_f
            if inicio_semana <= dt <= fim_semana:
                pago_semana += val_f
            if inicio_mes <= dt <= fim_mes:
                pago_mes += val_f

    # 2. Turnos operacionais encerrados (consumo estimado pela quilometragem)
    cur.execute("""
        SELECT id, data, km, ganho_bruto, ganho_dinheiro_pix, ganho_online,
               combustivel_consumido_estimado, combustivel, status_turno
        FROM turnos
        WHERE status_turno != 'aberto'
        ORDER BY data DESC, id DESC
    """)
    rows_turnos = cur.fetchall()
    consumo_hoje = 0.0
    consumo_semana = 0.0
    consumo_mes = 0.0
    km_hoje = 0.0
    km_semana = 0.0
    km_mes = 0.0
    bruto_hoje = 0.0
    bruto_semana = 0.0
    bruto_mes = 0.0

    for t in rows_turnos:
        t_id, dt, km, gb, g_din, g_onl, c_est, comb, st = t
        c_val = float(c_est or comb or 0.0)
        km_val = float(km or 0.0)
        gb_val = float(gb or 0.0)
        if dt == hoje_iso:
            consumo_hoje += c_val
            km_hoje += km_val
            bruto_hoje += gb_val
        if inicio_semana <= dt <= fim_semana:
            consumo_semana += c_val
            km_semana += km_val
            bruto_semana += gb_val
        if inicio_mes <= dt <= fim_mes:
            consumo_mes += c_val
            km_mes += km_val
            bruto_mes += gb_val

    # Resultados segregados (NUNCA desconta ambos simultaneamente)
    res_caixa_hoje = round(bruto_hoje - pago_hoje, 2)
    res_caixa_semana = round(bruto_semana - pago_semana, 2)
    res_caixa_mes = round(bruto_mes - pago_mes, 2)

    res_op_hoje = round(bruto_hoje - consumo_hoje, 2)
    res_op_semana = round(bruto_semana - consumo_semana, 2)
    res_op_mes = round(bruto_mes - consumo_mes, 2)

    return {
        "hoje": {
            "pago_real": round(pago_hoje, 2),
            "litros": round(litros_hoje, 2),
            "consumo_estimado": round(consumo_hoje, 2),
            "km_rodado": round(km_hoje, 1),
            "faturamento_bruto": round(bruto_hoje, 2),
            "resultado_caixa": res_caixa_hoje,
            "resultado_operacional_estimado": res_op_hoje
        },
        "semana": {
            "pago_real": round(pago_semana, 2),
            "litros": round(litros_semana, 2),
            "consumo_estimado": round(consumo_semana, 2),
            "km_rodado": round(km_semana, 1),
            "faturamento_bruto": round(bruto_semana, 2),
            "resultado_caixa": res_caixa_semana,
            "resultado_operacional_estimado": res_op_semana
        },
        "mes": {
            "pago_real": round(pago_mes, 2),
            "litros": round(litros_mes, 2),
            "consumo_estimado": round(consumo_mes, 2),
            "km_rodado": round(km_mes, 1),
            "faturamento_bruto": round(bruto_mes, 2),
            "resultado_caixa": res_caixa_mes,
            "resultado_operacional_estimado": res_op_mes
        },
        "abastecimentos_veiculo": abastecimentos,
        "despesas_sem_vinculo_veiculo": despesas_comb_avulsas
    }


def registrar_recebimento_turno(
    conn: sqlite3.Connection,
    turno_id: int,
    config_recebimento: Dict[str, Any]
) -> Tuple[bool, str]:
    """
    Contabiliza opcionalmente o recebimento financeiro dos ganhos de um turno.
    NUNCA cria transaÃ§Ãµes bancÃ¡rias automaticamente sem instruÃ§Ã£o explÃ­cita.
    Suporta:
    - modo 'nenhum': nÃ£o cria movimentaÃ§Ãµes bancÃ¡rias (deixa pendente).
    - modo 'conta_unica': deposita valor na conta especificada.
    - modo 'divisao': reparte valor entre mÃºltiplas contas.
    - modo 'vincular': vincula a uma transaÃ§Ã£o de receita prÃ©-existente (sem duplicaÃ§Ã£o).
    - crÃ©dito de ganhos online quando confirmado.
    """
    cur = conn.cursor()
    cur.execute("SELECT id, data, ganho_dinheiro_pix, ganho_online, ganho_bruto FROM turnos WHERE id = ?", (turno_id,))
    t_row = cur.fetchone()
    if not t_row:
        return False, "Turno nÃ£o encontrado."

    dt_turno = t_row[1]
    modo = config_recebimento.get("modo", "nenhum")
    mensagens = []

    # 1. Tratamento de Dinheiro / Pix
    if modo == "conta_unica":
        conta_id = config_recebimento.get("conta_id")
        valor = float(config_recebimento.get("valor", t_row[2]))
        if valor > 0 and conta_id:
            cur.execute("SELECT nome FROM contas_config WHERE id = ?", (conta_id,))
            c_row = cur.fetchone()
            c_nome = c_row[0] if c_row else "Conta"
            cur.execute("""
                INSERT INTO transacoes (data, descricao, valor, tipo, categoria, conta, conta_id, pago, afeta_saldo_bancario, turno_id)
                VALUES (?, ?, ?, 'receita', '99 / Chaves (Carlos)', ?, ?, 1, 1, ?)
            """, (dt_turno, f"Ganhos 99 - Dinheiro/Pix (Turno #{turno_id})", valor, c_nome, conta_id, turno_id))
            mensagens.append(f"R$ {valor:,.2f} em {c_nome}")

    elif modo == "divisao":
        divisoes = config_recebimento.get("divisoes", [])
        for div in divisoes:
            c_id = div.get("conta_id")
            val = float(div.get("valor", 0.0))
            if val > 0 and c_id:
                cur.execute("SELECT nome FROM contas_config WHERE id = ?", (c_id,))
                c_row = cur.fetchone()
                c_nome = c_row[0] if c_row else "Conta"
                cur.execute("""
                    INSERT INTO transacoes (data, descricao, valor, tipo, categoria, conta, conta_id, pago, afeta_saldo_bancario, turno_id)
                    VALUES (?, ?, ?, 'receita', '99 / Chaves (Carlos)', ?, ?, 1, 1, ?)
                """, (dt_turno, f"Ganhos 99 - Dinheiro/Pix (Turno #{turno_id}) - {c_nome}", val, c_nome, c_id, turno_id))
                mensagens.append(f"R$ {val:,.2f} em {c_nome}")

    elif modo == "vincular":
        transacao_id = config_recebimento.get("transacao_id")
        if transacao_id:
            cur.execute("UPDATE transacoes SET turno_id = ? WHERE id = ?", (turno_id, transacao_id))
            cur.execute("SELECT descricao, valor, conta FROM transacoes WHERE id = ?", (transacao_id,))
            tr_vinc = cur.fetchone()
            if tr_vinc:
                mensagens.append(f"Vinculado Ã  entrada '{tr_vinc[0]}' (R$ {tr_vinc[1]:,.2f} em {tr_vinc[2]})")

    # 2. Tratamento de Ganhos Online
    if config_recebimento.get("online_creditado", False):
        onl_conta_id = config_recebimento.get("online_conta_id")
        onl_valor = float(config_recebimento.get("online_valor", t_row[3]))
        if onl_valor > 0 and onl_conta_id:
            cur.execute("SELECT nome FROM contas_config WHERE id = ?", (onl_conta_id,))
            c_row = cur.fetchone()
            c_nome = c_row[0] if c_row else "Conta"
            cur.execute("""
                INSERT INTO transacoes (data, descricao, valor, tipo, categoria, conta, conta_id, pago, afeta_saldo_bancario, turno_id)
                VALUES (?, ?, ?, 'receita', '99 / Chaves (Carlos)', ?, ?, 1, 1, ?)
            """, (dt_turno, f"Repasse/CrÃ©dito Online 99 (Turno #{turno_id})", onl_valor, c_nome, onl_conta_id, turno_id))
            mensagens.append(f"Online: R$ {onl_valor:,.2f} em {c_nome}")

    conn.commit()
    if mensagens:
        return True, "Recebimento registrado: " + " â€¢ ".join(mensagens)
    return True, "Turno gravado sem movimentaÃ§Ãµes financeiras imediatas (Pendente no extrato)."

def transferir_dinheiro_para_mercado_pago(

    data_str: str,

) -> Tuple[bool, str]:
    """
    Registra transferÃªncia interna de 'Dinheiro / Carteira' para 'Mercado Pago'.

        VALUES (?, ?, ?, 'transferencia', 'TransferÃªncia', ?, ?, ?, 1, 1)
    """, (data_str, desc.strip(), float(valor), cart_nome, cart_id, mp_id))
    conn.commit()
    return True, f"TransferÃªncia de R$ {valor:,.2f} ({cart_nome} âž” {mp_nome}) registrada com sucesso sem gerar receita duplicada!"

def salvar_abastecimento_veiculo(
    conn: sqlite3.Connection,
    dados: Dict[str, Any]
) -> Tuple[bool, str]:
    """
    Salva ou atualiza um abastecimento no mÃ³dulo do veÃ­culo.
    Se dados['transacao_id'] for informado, VINCULA Ã  transaÃ§Ã£o existente SEM criar nova saÃ­da financeira.
    Se dados['transacao_id'] nÃ£o for informado, cria um ÃšNICO lanÃ§amento financeiro em transacoes com valor = valor_pago.
    """
    cur = conn.cursor()
    data_str = dados.get("data") or str(date.today())
    hodometro = float(dados.get("hodometro") or 0.0)
    combustivel = dados.get("combustivel") or "Etanol"
    litros = float(dados.get("litros") or 0.0)
    valor_bruto = float(dados.get("valor_bruto") or 0.0)
    desconto = float(dados.get("desconto") or 0.0)
    valor_pago = float(dados.get("valor_pago") or max(0.0, valor_bruto - desconto))
    conta = dados.get("conta") or "Dinheiro / Carteira"
    conta_id = dados.get("conta_id") or 2
    tanque_cheio = 1 if dados.get("tanque_cheio", True) else 0
    posto = (dados.get("posto") or "").strip()
    obs = (dados.get("observacao") or "").strip()
    transacao_id = dados.get("transacao_id")

    preco_litro = round(valor_bruto / litros, 3) if (litros and litros > 0) else None

    # Caso 0: AtualizaÃ§Ã£o de registro existente de veiculo_abastecimentos
    id_abast = dados.get("id")
    if id_abast:
        cur.execute("""
            UPDATE veiculo_abastecimentos
            SET data = ?, hodometro = ?, combustivel = ?, litros = ?, preco_litro = ?,
                valor_bruto = ?, desconto = ?, valor_pago = ?, conta = ?, conta_id = ?,
                tanque_cheio = ?, posto = ?, observacao = ?
            WHERE id = ?
        """, (data_str, hodometro, combustivel, litros, preco_litro, valor_bruto,
              desconto, valor_pago, conta, conta_id, tanque_cheio, posto, obs, id_abast))
        conn.commit()
        return True, "Registro de abastecimento atualizado com sucesso!"

    # Caso 1: JÃ¡ possui transacao_id vinculada (ou fornecida pelo usuÃ¡rio)
    if transacao_id:
        cur.execute("SELECT id, valor, conta FROM transacoes WHERE id = ?", (transacao_id,))
        r_tr = cur.fetchone()
        if not r_tr:
            return False, f"TransaÃ§Ã£o #{transacao_id} nÃ£o encontrada no extrato financeiro."
        
        # Se jÃ¡ existir vÃ­nculo para esta transaÃ§Ã£o, atualiza os dados em vez de duplicar
        cur.execute("SELECT id FROM veiculo_abastecimentos WHERE transacao_id = ?", (transacao_id,))
        r_exist = cur.fetchone()
        if r_exist:
            cur.execute("""
                UPDATE veiculo_abastecimentos
                SET data = ?, hodometro = ?, combustivel = ?, litros = ?, preco_litro = ?,
                    valor_bruto = ?, desconto = ?, valor_pago = ?, conta = ?, conta_id = ?,
                    tanque_cheio = ?, posto = ?, observacao = ?
                WHERE id = ?
            """, (data_str, hodometro, combustivel, litros, preco_litro, valor_bruto,
                  desconto, valor_pago, conta, conta_id, tanque_cheio, posto, obs, r_exist[0]))
            conn.commit()
            return True, f"Abastecimento #{r_exist[0]} vinculado Ã  Despesa #{transacao_id} atualizado com sucesso!"

        # Insere em veiculo_abastecimentos vinculado
        cur.execute("""
            INSERT INTO veiculo_abastecimentos (
                data, hodometro, combustivel, litros, preco_litro, valor_bruto,
                desconto, valor_pago, conta, conta_id, tanque_cheio, posto, observacao, transacao_id
            ) VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)
        """, (data_str, hodometro, combustivel, litros, preco_litro, valor_bruto,
              desconto, valor_pago, conta, conta_id, tanque_cheio, posto, obs, transacao_id))
        conn.commit()
        return True, f"Abastecimento ({combustivel} â€¢ R$ {valor_pago:,.2f}) vinculado com sucesso Ã  Despesa #{transacao_id} sem duplicar saÃ­da bancÃ¡ria!"

    # Caso 2: Novo abastecimento com criaÃ§Ã£o de despesa Ãºnica
    else:
        desc_tr = f"Abastecimento {combustivel}" + (f" ({litros:.1f}L)" if (litros and litros > 0) else "") + (f" - {posto}" if posto else "")
        cur.execute("""
            INSERT INTO transacoes (data, descricao, valor, tipo, categoria, conta, conta_id, pago, afeta_saldo_bancario)
            VALUES (?, ?, ?, 'despesa', 'CombustÃ­vel (Trabalho)', ?, ?, 1, 1)
        """, (data_str, desc_tr, valor_pago, conta, conta_id))
        nova_transacao_id = cur.lastrowid

        cur.execute("""
            INSERT INTO veiculo_abastecimentos (
                data, hodometro, combustivel, litros, preco_litro, valor_bruto,
                desconto, valor_pago, conta, conta_id, tanque_cheio, posto, observacao, transacao_id
            ) VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)
        """, (data_str, hodometro, combustivel, litros, preco_litro, valor_bruto,
              desconto, valor_pago, conta, conta_id, tanque_cheio, posto, obs, nova_transacao_id))
        conn.commit()
        return True, f"Abastecimento de R$ {valor_pago:,.2f} ({combustivel}) registrado com sucesso no veÃ­culo e no extrato ({conta})!"


def excluir_abastecimento_veiculo(
    conn: sqlite3.Connection,
    id_abast: int,
    excluir_transacao_financeira: bool = False
) -> Tuple[bool, str]:
    """Remove o registro do veÃ­culo. Se excluir_transacao_financeira for True, tambÃ©m remove o lanÃ§amento em transacoes."""
    cur = conn.cursor()
    cur.execute("SELECT id, transacao_id, valor_pago, combustivel FROM veiculo_abastecimentos WHERE id = ?", (id_abast,))
    row = cur.fetchone()
    if not row:
        return False, "Abastecimento nÃ£o encontrado."

    tr_id = row[1]
    val = row[2]
    comb = row[3]

    cur.execute("DELETE FROM veiculo_abastecimentos WHERE id = ?", (id_abast,))
    if excluir_transacao_financeira and tr_id:
        cur.execute("DELETE FROM transacoes WHERE id = ?", (tr_id,))

    conn.commit()
    msg = f"Abastecimento ({comb} â€¢ R$ {val:,.2f}) removido do veÃ­culo"
    if excluir_transacao_financeira and tr_id:
        msg += " e despesa financeira removida do extrato."
    else:
        msg += " (lanÃ§amento financeiro preservado no extrato)."
    return True, msg

CENTAVOS = Decimal("0.01")


def to_decimal(val: Any) -> Decimal:
    """Converte valor para Decimal de 2 casas decimais de forma segura."""
    if val is None:
        return Decimal("0.00")
    if isinstance(val, Decimal):
        return val.quantize(CENTAVOS, rounding=ROUND_HALF_EVEN)
    try:
        return Decimal(str(val)).quantize(CENTAVOS, rounding=ROUND_HALF_EVEN)
    except Exception:
        return Decimal("0.00")


def inicializar_esquema_planejamento(conn: sqlite3.Connection) -> None:
    """
    Cria as tabelas necessÃ¡rias para o mÃ³dulo de planejamento financeiro inteligente
    sem modificar nem afetar as tabelas prÃ©-existentes.
    """
    cur = conn.cursor()

    # 1. Metas especÃ­ficas e MissÃ£o PrioritÃ¡ria (ex: IPVA)
    cur.execute("""
    CREATE TABLE IF NOT EXISTS planejamento_metas (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        mes_ano TEXT NOT NULL,
        titulo TEXT NOT NULL,
        transacao_alvo_id INTEGER REFERENCES transacoes(id),
        valor_alvo REAL NOT NULL,
        valor_reservado REAL DEFAULT 0.0,
        data_limite DATE NOT NULL,
        turnos_planejados INTEGER DEFAULT 5,
        horas_por_turno REAL DEFAULT 4.0,
        ganho_bruto_hora_estimado REAL DEFAULT 35.0,
        combustivel_hora_estimado REAL DEFAULT 10.0,
        observacoes TEXT,
        criado_em TEXT DEFAULT CURRENT_TIMESTAMP
    );
    """)

    # 2. PrevisÃ£o de despesas variÃ¡veis realistas (a partir de hoje para outubro, mensal para seguintes)
    cur.execute("""
    CREATE TABLE IF NOT EXISTS planejamento_previsao_variaveis (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        mes_ano TEXT NOT NULL,
        categoria TEXT NOT NULL,
        tipo_previsao TEXT DEFAULT 'mensal',
        data_inicio DATE,
        valor_previsto REAL NOT NULL,
        essencial_semanal REAL DEFAULT 0.0,
        observacoes TEXT,
        UNIQUE(mes_ano, categoria)
    );
    """)

    # 3. PriorizaÃ§Ã£o de contas atrasadas por semana de referÃªncia
    cur.execute("""
    CREATE TABLE IF NOT EXISTS planejamento_prioridades_atrasados (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        semana_ref TEXT NOT NULL,
        transacao_id INTEGER NOT NULL REFERENCES transacoes(id),
        prioritaria INTEGER DEFAULT 1,
        valor_alocado REAL DEFAULT 0.0,
        UNIQUE(semana_ref, transacao_id)
    );
    """)

    # 4. ParÃ¢metros de simulaÃ§Ã£o de capacidade de trabalho do motorista
    cur.execute("""
    CREATE TABLE IF NOT EXISTS planejamento_simulacao_trabalho (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        semana_ref TEXT NOT NULL UNIQUE,
        dias_disponiveis INTEGER DEFAULT 5,
        horas_disponiveis_dia REAL DEFAULT 4.0,
        ganho_bruto_hora_estimado REAL DEFAULT 35.0,
        combustivel_hora_estimado REAL DEFAULT 10.0,
        observacoes TEXT
    );
    """)

    # 5. Adiciona coluna auxiliar em transacoes se nÃ£o existir para sinalizar faturas sob conciliaÃ§Ã£o
    try:
        cur.execute("ALTER TABLE transacoes ADD COLUMN status_conciliacao_fatura TEXT DEFAULT 'normal'")
    except Exception:
        pass

    # 6. Adiciona status e prioridade em planejamento_metas para missÃµes flexÃ­veis
    cur.execute("PRAGMA table_info(planejamento_metas)")
    cols_metas = [c[1] for c in cur.fetchall()]
    if 'status' not in cols_metas:
        try:
            cur.execute("ALTER TABLE planejamento_metas ADD COLUMN status TEXT DEFAULT 'ativa'")
        except Exception:
            pass
    if 'prioridade' not in cols_metas:
        try:
            cur.execute("ALTER TABLE planejamento_metas ADD COLUMN prioridade TEXT DEFAULT 'alta'")
        except Exception:
            pass

    # 7. Tabela de Cofrinhos Reais (Mercado Pago / Envelopes)
    cur.execute("""
    CREATE TABLE IF NOT EXISTS cofrinhos (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        conta_pai_id INTEGER DEFAULT 1 REFERENCES contas_config(id),
        nome TEXT NOT NULL UNIQUE,
        saldo_confirmado REAL DEFAULT 0.0,
        meta_valor REAL DEFAULT 0.0,
        objetivo TEXT,
        data_atualizacao DATE,
        observacoes TEXT,
        criado_em TIMESTAMP DEFAULT CURRENT_TIMESTAMP
    );
    """)

    # Sementes dos cofrinhos reais conhecidos pelo usuÃ¡rio (iniciados com saldo R$ 0,00 sem presumir positivo)
    cofrinhos_iniciais = [
        ("Cofrinho IPVA", "Reserva para IPVA 2025/2026", 431.11),
        ("Cofrinho Acordo BV", "Reserva para Acordo BV", 1428.82),
        ("Cofrinho Rubens", "Reserva Rubens", 0.0),
        ("Cofrinho Contas de Casa", "Reserva para contas essenciais da casa", 0.0),
        ("Cofrinho Mercado", "AlimentaÃ§Ã£o e supermercado", 0.0),
        ("Cofrinho Reserva de SeguranÃ§a", "Fundo de emergÃªncia familiar", 0.0)
    ]
    for nome_c, obj_c, meta_c in cofrinhos_iniciais:
        cur.execute("""
            INSERT OR IGNORE INTO cofrinhos (conta_pai_id, nome, saldo_confirmado, meta_valor, objetivo, data_atualizacao)
            VALUES (1, ?, 0.0, ?, ?, date('now'))
        """, (nome_c, meta_c, obj_c))

    # Garante colunas de conta vinculada e inclusÃ£o no consolidado em cofrinhos
    cur.execute("PRAGMA table_info(cofrinhos)")
    cols_cof = [r[1] for r in cur.fetchall()]
    if "conta_config_id" not in cols_cof:
        try:
            cur.execute("ALTER TABLE cofrinhos ADD COLUMN conta_config_id INTEGER REFERENCES contas_config(id)")
        except Exception:
            pass
    if "incluir_saldo_consolidado" not in cols_cof:
        try:
            cur.execute("ALTER TABLE cofrinhos ADD COLUMN incluir_saldo_consolidado INTEGER DEFAULT 0")
        except Exception:
            pass

    conn.commit()

# 1. CÃLCULO DA NECESSIDADE DE CAIXA DOS PRÃ“XIMOS 7 DIAS
# =============================================================================
def calcular_necessidade_caixa_7dias(
    conn: sqlite3.Connection,
    data_referencia: Optional[date] = None,
    semana_ref: Optional[str] = None
) -> Dict[str, Any]:
    """
    Calcula a necessidade de caixa dos prÃ³ximos 7 dias diferenciando:
    - ObrigaÃ§Ãµes confirmadas da semana.
    - ObrigaÃ§Ãµes vencidas selecionadas como prioritÃ¡rias para esta semana.
    - ObrigaÃ§Ãµes vencidas reprogramadas (nÃ£o somadas na meta imediata).
    - Valores pendentes de confirmaÃ§Ã£o (ex: Acordo BV de R$ 1.428,82).
    - Gastos variÃ¡veis essenciais esperados.
    - Saldo bancÃ¡rio total, reservas virtuais e saldo livre.
    - Entradas confirmadas e rendas estimadas.
    """
    if data_referencia is None:
        data_referencia = date.today()
    if semana_ref is None:
        semana_ref = data_referencia.strftime("%Y-W%W")

    data_limite_7d = data_referencia + timedelta(days=6)
    dt_ref_str = data_referencia.strftime("%Y-%m-%d")
    dt_limite_str = data_limite_7d.strftime("%Y-%m-%d")
    mes_ano_ref = data_referencia.strftime("%Y-%m")

    cur = conn.cursor()

    # 1. Saldo BancÃ¡rio Consolidado e Reservas Virtuais
    saldo_info = saldo_service.calcular_saldo_consolidado(conn, data_referencia)
    saldo_total_disponivel = saldo_info["saldo_consolidado"]

    # Busca total de reservas virtuais alocadas em planejamento_metas
    cur.execute("""
        SELECT COALESCE(SUM(valor_reservado), 0.0)
        FROM planejamento_metas
        WHERE mes_ano = ?
    """, (mes_ano_ref,))
    res_row = cur.fetchone()
    total_reservas_virtuais = to_decimal(res_row[0] if res_row else 0.0)

    # Saldo livre para outros compromissos (nunca negativo para fins de cÃ¡lculo de caixa)
    saldo_livre = max(Decimal("0.00"), saldo_total_disponivel - total_reservas_virtuais)

    # 2. Busca lista de transaÃ§Ãµes priorizadas para esta semana
    cur.execute("""
        SELECT transacao_id, prioritaria, valor_alocado
        FROM planejamento_prioridades_atrasados
        WHERE semana_ref = ?
    """, (semana_ref,))
    mapa_prioridades = {r[0]: (r[1], r[2]) for r in cur.fetchall()}

    # 3. Mapeamento de todas as obrigaÃ§Ãµes pendentes (tipo='despesa', pago != 1)
    cur.execute("""
        SELECT t.id, t.data, t.descricao, t.valor, t.categoria, t.conta, t.pago,
               COALESCE(t.status_conciliacao_fatura, 'normal') as status_fatura
        FROM transacoes t
        WHERE t.tipo = 'despesa' AND (t.pago IS NULL OR t.pago != 1)
        ORDER BY t.data ASC, t.id ASC
    """)
    linhas_despesas = cur.fetchall()

    obrigacoes_confirmadas_7d = []
    obrigacoes_vencidas_prioritarias = []
    obrigacoes_vencidas_reprogramadas = []
    valores_pendentes_confirmacao = []

    total_confirmadas_7d = Decimal("0.00")
    total_vencidas_prioritarias = Decimal("0.00")
    total_vencidas_reprogramadas = Decimal("0.00")
    total_pendente_confirmacao = Decimal("0.00")

    for r in linhas_despesas:
        t_id, t_data, t_desc, t_valor_nom, t_cat, t_conta, t_pago, t_st_fatura = r
        dt_venc = datetime.strptime(t_data, "%Y-%m-%d").date() if isinstance(t_data, str) else t_data

        # ObtÃ©m saldo devedor real descontando pagamentos parciais jÃ¡ feitos
        resumo_obrig = saldo_service.obter_resumo_obrigacao(conn, t_id)
        saldo_devedor = resumo_obrig["saldo_devedor"]

        if saldo_devedor <= Decimal("0.00"):
            continue

        item_dict = {
            "id": t_id,
            "data": t_data,
            "descricao": t_desc,
            "valor_nominal": to_decimal(t_valor_nom),
            "saldo_devedor": saldo_devedor,
            "categoria": t_cat,
            "conta": t_conta,
            "status_fatura": t_st_fatura
        }

        # Caso A: Fatura ou acordo com parcelamento pendente de confirmaÃ§Ã£o (ex: BV)
        if t_st_fatura == "pendente_confirmacao" or "Acordo BV" in t_desc:
            valores_pendentes_confirmacao.append(item_dict)
            total_pendente_confirmacao += saldo_devedor
            continue

        # Caso B: Conta Vencida antes da data de referÃªncia
        if dt_venc < data_referencia:
            is_prioritaria = mapa_prioridades.get(t_id, (0, 0.0))[0] == 1
            if is_prioritaria:
                val_alocado = to_decimal(mapa_prioridades[t_id][1])
                val_efetivo = val_alocado if val_alocado > 0 else saldo_devedor
                item_dict["valor_priorizado"] = val_efetivo
                obrigacoes_vencidas_prioritarias.append(item_dict)
                total_vencidas_prioritarias += val_efetivo
            else:
                obrigacoes_vencidas_reprogramadas.append(item_dict)
                total_vencidas_reprogramadas += saldo_devedor

        # Caso C: Conta Vincenda nos prÃ³ximos 7 dias (data_ref <= venc <= data_limite_7d)
        elif data_referencia <= dt_venc <= data_limite_7d:
            obrigacoes_confirmadas_7d.append(item_dict)
            total_confirmadas_7d += saldo_devedor

    # 4. Gastos VariÃ¡veis Essenciais Esperados na Semana
    cur.execute("""
        SELECT COALESCE(SUM(essencial_semanal), 0.0)
        FROM planejamento_previsao_variaveis
        WHERE mes_ano = ?
    """, (mes_ano_ref,))
    row_var = cur.fetchone()
    gastos_variaveis_semana = to_decimal(row_var[0] if row_var else Decimal("150.00"))
    if gastos_variaveis_semana == Decimal("0.00"):
        # ReferÃªncia padrÃ£o conservadora para despesas essenciais bÃ¡sicas (Supermercado + FarmÃ¡cia)
        gastos_variaveis_semana = Decimal("150.00")

    # 5. Entradas Futuras Confirmadas nos PrÃ³ximos 7 Dias (receitas pendentes confirmadas)
    cur.execute("""
        SELECT COALESCE(SUM(valor), 0.0)
        FROM transacoes
        WHERE tipo = 'receita' AND pago = 0 AND data >= ? AND data <= ?
    """, (dt_ref_str, dt_limite_str))
    row_rec = cur.fetchone()
    entradas_futuras_confirmadas = to_decimal(row_rec[0] if row_rec else 0.0)

    # 6. CÃ¡lculo dos Totais e da Necessidade Adicional de Caixa
    compromissos_minimos_semana = (
        total_confirmadas_7d +
        total_vencidas_prioritarias +
        gastos_variaveis_semana
    )

    # A necessidade adicional Ã© o que precisa entrar de dinheiro novo
    necessidade_adicional_caixa = max(
        Decimal("0.00"),
        compromissos_minimos_semana - saldo_livre - entradas_futuras_confirmadas
    )

    return {
        "data_referencia": dt_ref_str,
        "data_limite_7d": dt_limite_str,
        "saldo_total_disponivel": saldo_total_disponivel,
        "total_reservas_virtuais": total_reservas_virtuais,
        "saldo_livre": saldo_livre,
        "obrigacoes_confirmadas_7d": obrigacoes_confirmadas_7d,
        "total_confirmadas_7d": total_confirmadas_7d,
        "obrigacoes_vencidas_prioritarias": obrigacoes_vencidas_prioritarias,
        "total_vencidas_prioritarias": total_vencidas_prioritarias,
        "obrigacoes_vencidas_reprogramadas": obrigacoes_vencidas_reprogramadas,
        "total_vencidas_reprogramadas": total_vencidas_reprogramadas,
        "valores_pendentes_confirmacao": valores_pendentes_confirmacao,
        "total_pendente_confirmacao": total_pendente_confirmacao,
        "gastos_variaveis_semana": gastos_variaveis_semana,
        "entradas_futuras_confirmadas": entradas_futuras_confirmadas,
        "compromissos_minimos_semana": compromissos_minimos_semana,
        "necessidade_adicional_caixa": necessidade_adicional_caixa,
        "status_bv_pendente": len(valores_pendentes_confirmacao) > 0,
        "aviso_bv": (
            f"Existe R$ {total_pendente_confirmacao:,.2f} em parcelamento de fatura pendente de conciliaÃ§Ã£o (BV). "
            f"A necessidade calculada de R$ {necessidade_adicional_caixa:,.2f} representa apenas o mÃ­nimo confirmado."
            if total_pendente_confirmacao > 0 else ""
        )
    }

# 2. CÃLCULO DA PROJEÃ‡ÃƒO MENSAL (FECHAMENTO ATÃ‰ O FIM DO MÃŠS)
# =============================================================================
def calcular_projecao_mensal(
    conn: sqlite3.Connection,
    data_referencia: Optional[date] = None
) -> Dict[str, Any]:
    """
    Projeta o fechamento do mÃªs (ex: 31/10/2026) considerando:
    - Todas as obrigaÃ§Ãµes pendentes atÃ© o final do mÃªs.
    - PrevisÃ£o de despesas variÃ¡veis manuais a partir da data atual (para outubro).
    - Desconto de valores reservados virtualmente para missÃµes (IPVA).
    - Saldo final projetado e dÃ©ficit/superÃ¡vit.
    """
    if data_referencia is None:
        data_referencia = date.today()

    dt_ref_str = data_referencia.strftime("%Y-%m-%d")
    mes_ano_ref = data_referencia.strftime("%Y-%m")
    ano = data_referencia.year
    mes = data_referencia.month
    
    # Ãšltimo dia do mÃªs
    if mes == 12:
        ultimo_dia = date(ano, 12, 31)
    else:
        ultimo_dia = date(ano, mes + 1, 1) - timedelta(days=1)
    ultimo_dia_str = ultimo_dia.strftime("%Y-%m-%d")

    cur = conn.cursor()

    # 1. Saldo inicial disponÃ­vel
    saldo_info = saldo_service.calcular_saldo_consolidado(conn, data_referencia)
    saldo_total_disponivel = saldo_info["saldo_consolidado"]

    # 2. Despesas fixas/obrigaÃ§Ãµes pendentes no mÃªs
    cur.execute("""
        SELECT t.id, t.valor, COALESCE(t.status_conciliacao_fatura, 'normal')
        FROM transacoes t
        WHERE t.tipo = 'despesa' AND (t.pago IS NULL OR t.pago != 1)
          AND t.data <= ?
    """, (ultimo_dia_str,))
    rows_obrig = cur.fetchall()

    total_obrigacoes_confirmadas = Decimal("0.00")
    total_bv_pendente = Decimal("0.00")

    for r in rows_obrig:
        t_id, t_val, t_st = r
        resumo = saldo_service.obter_resumo_obrigacao(conn, t_id)
        saldo_dev = resumo["saldo_devedor"]
        if saldo_dev <= Decimal("0.00"):
            continue
        if t_st == "pendente_confirmacao":
            total_bv_pendente += saldo_dev
        else:
            total_obrigacoes_confirmadas += saldo_dev

    # 3. Despesas VariÃ¡veis Restantes (modelo manual a partir de hoje para outubro)
    resumo_variaveis = calcular_despesas_variaveis_restantes(conn, mes_ano_ref, data_referencia)
    total_variaveis_restantes = resumo_variaveis["total_consumo_esperado_restante"]

    # 4. Total de reservas alocadas para missÃµes
    cur.execute("""
        SELECT COALESCE(SUM(valor_reservado), 0.0)
        FROM planejamento_metas
        WHERE mes_ano = ?
    """, (mes_ano_ref,))
    row_res = cur.fetchone()
    total_reservas = to_decimal(row_res[0] if row_res else 0.0)

    # 5. Receitas confirmadas previstas atÃ© o fim do mÃªs
    cur.execute("""
        SELECT COALESCE(SUM(valor), 0.0)
        FROM transacoes
        WHERE tipo = 'receita' AND pago = 0 AND data >= ? AND data <= ?
    """, (dt_ref_str, ultimo_dia_str))
    row_rec = cur.fetchone()
    receitas_confirmadas_previstas = to_decimal(row_rec[0] if row_rec else 0.0)

    # Total de saÃ­das mÃ­nimas confirmadas
    total_saidas_projetadas = total_obrigacoes_confirmadas + total_variaveis_restantes

    # Saldo final projetado (sem contar rendas futuras ainda nÃ£o trabalhadas)
    saldo_final_projetado_sem_renda_app = (
        saldo_total_disponivel +
        receitas_confirmadas_previstas -
        total_saidas_projetadas
    )

    necessidade_geral_fim_mes = max(Decimal("0.00"), -saldo_final_projetado_sem_renda_app)

    return {
        "mes_ano": mes_ano_ref,
        "ultimo_dia_mes": ultimo_dia_str,
        "saldo_atual": saldo_total_disponivel,
        "total_reservas_virtuais": total_reservas,
        "total_obrigacoes_confirmadas": total_obrigacoes_confirmadas,
        "total_bv_pendente": total_bv_pendente,
        "total_variaveis_restantes": total_variaveis_restantes,
        "receitas_confirmadas_previstas": receitas_confirmadas_previstas,
        "total_saidas_projetadas": total_saidas_projetadas,
        "saldo_final_projetado_sem_renda_app": saldo_final_projetado_sem_renda_app,
        "necessidade_geral_fim_mes": necessidade_geral_fim_mes,
        "resumo_variaveis": resumo_variaveis
    }

# 3. GESTÃƒO DE DESPESAS VARIÃVEIS (PREVISÃƒO A PARTIR DE HOJE EM OUTUBRO)
# =============================================================================
def calcular_despesas_variaveis_restantes(
    conn: sqlite3.Connection,
    mes_ano: Optional[str] = None,
    data_referencia: Optional[date] = None
) -> Dict[str, Any]:
    """
    Calcula as despesas variÃ¡veis esperadas.
    Para outubro/2026:
    Utiliza previsÃ£o manual de gastos futuros a partir da data atual (19/10 a 31/10).
    LanÃ§amentos reais POSTERIORES Ã  data de referÃªncia consomem e abatem a previsÃ£o,
    sem duplicaÃ§Ã£o de valores.
    Exclui rigorosamente dÃ­vidas, acordos, cartÃµes e parcelas.
    """
    if data_referencia is None:
        data_referencia = date.today()
    if mes_ano is None:
        mes_ano = data_referencia.strftime("%Y-%m")

    cur = conn.cursor()
    dt_ref_str = data_referencia.strftime("%Y-%m-%d")

    # Categorias de dÃ­vidas/parcelas e despesas contratuais fixas a ignorar estritamente
    # para evitar duplicidade com as obrigaÃ§Ãµes confirmadas de contas a pagar
    CATEGORIAS_IGNORAR = {
        "Acordo BV", "Outras DÃ­vidas", "Pagamento de fatura", "Impostos VeÃ­culo",
        "Saldo Inicial", "SalÃ¡rio Fernanda", "Renda Corridas (99/Uber)",
        "HabitaÃ§Ã£o", "Ãgua (Copasa)", "Energia (CEMIG + CEMIG SIM)", "Internet Residencial",
        "Telefone (Fernanda)", "Streaming", "Streaming / Entretenimento", "Plano de SaÃºde",
        "Plano Odonto", "PÃ³s-GraduaÃ§Ã£o Fernanda", "Seguro VeÃ­culo (HB20 AGV)",
        "DocumentaÃ§Ã£o (IPVA, Multas)", "Conta Capital Sicoob", "Crediserv - EmprÃ©stimo",
        "DÃ­zimo (Igreja)", "Reserva de EmergÃªncia"
    }

    # Busca limites de gastos cadastrados (tetos de seguranÃ§a)
    cur.execute("SELECT categoria, limite FROM limites_categoria")
    tetos_categoria = {r[0]: to_decimal(r[1]) for r in cur.fetchall()}

    # Busca previsÃµes manuais cadastradas para o mÃªs
    cur.execute("""
        SELECT categoria, valor_previsto, essencial_semanal, tipo_previsao
        FROM planejamento_previsao_variaveis
        WHERE mes_ano = ?
    """, (mes_ano,))
    previsoes_db = {r[0]: (to_decimal(r[1]), to_decimal(r[2]), r[3]) for r in cur.fetchall()}

    # Valores padrÃ£o iniciais de previsÃ£o manual a partir de 19/10 (para quando nÃ£o houver registro manual)
    PREVISOES_PADRAO_OUTUBRO = {
        "Supermercado": (Decimal("350.00"), Decimal("120.00")),
        "Medicamentos": (Decimal("80.00"), Decimal("30.00")),
        "CombustÃ­vel": (Decimal("200.00"), Decimal("70.00")),
        "Restaurante": (Decimal("100.00"), Decimal("0.00")),
        "Outros": (Decimal("60.00"), Decimal("0.00"))
    }

    detalhe_categorias = []
    total_consumo_esperado_restante = Decimal("0.00")
    total_essencial_semana = Decimal("0.00")

    todas_categorias = set(tetos_categoria.keys()) | set(previsoes_db.keys()) | set(PREVISOES_PADRAO_OUTUBRO.keys())
    todas_categorias = {c for c in todas_categorias if c not in CATEGORIAS_IGNORAR}

    for cat in sorted(todas_categorias):
        teto = tetos_categoria.get(cat, Decimal("0.00"))

        if cat in previsoes_db:
            val_previsto, val_essencial_sem, tipo_prev = previsoes_db[cat]
        elif mes_ano == "2026-10" and cat in PREVISOES_PADRAO_OUTUBRO:
            val_previsto, val_essencial_sem = PREVISOES_PADRAO_OUTUBRO[cat]
            tipo_prev = "a_partir_de_hoje"
        elif any(k in cat.lower() for k in ["supermercado", "alimenta", "refei", "padaria", "farm", "medic", "combust", "restaurante", "lazer", "eventual"]):

        else:
            val_previsto = Decimal("0.00")
            val_essencial_sem = Decimal("0.00")
            tipo_prev = "ignorado_fixo"

        # Gasto jÃ¡ realizado DEPOIS da data de referÃªncia (apenas despesas avulsas pagas, sem parcelas futuras)
        cur.execute("""
            SELECT COALESCE(SUM(valor), 0.0)
            FROM transacoes
            WHERE (categoria = ? OR subcategoria_id IN (SELECT id FROM subcategorias WHERE nome = ?))

        # Consumo restante esperado: a despesa real posterior substitui a previsÃ£o
        consumo_restante = max(Decimal("0.00"), val_previsto - gasto_posterior)
        total_consumo_esperado_restante += consumo_restante
        total_essencial_semana += val_essencial_sem

        detalhe_categorias.append({
            "categoria": cat,
            "teto_limite": teto,
            "previsao_futura": val_previsto,
            "gasto_posterior_realizado": gasto_posterior,
            "consumo_restante": consumo_restante,
            "essencial_semanal": val_essencial_sem,
            "tipo_previsao": tipo_prev
        })

    return {
        "mes_ano": mes_ano,
        "detalhe_categorias": detalhe_categorias,
        "total_consumo_esperado_restante": total_consumo_esperado_restante,
        "total_essencial_semana": total_essencial_semana
    }

# 4. MISSÃƒO PRIORITÃRIA (IPVA) E GESTÃƒO DE RESERVAS VIRTUAIS
# =============================================================================
def obter_status_missao_ipva(
    conn: sqlite3.Connection,
    mes_ano: Optional[str] = None,
    data_referencia: Optional[date] = None
) -> Dict[str, Any]:
    """
    Recupera a MissÃ£o PrioritÃ¡ria vinculada Ã  obrigaÃ§Ã£o existente do IPVA (id=33),
    sem criar duplicidade de passivo nem gerar movimentaÃ§Ãµes bancÃ¡rias automÃ¡ticas.
    """
    if data_referencia is None:
        data_referencia = date.today()
    if mes_ano is None:
        mes_ano = data_referencia.strftime("%Y-%m")

    cur = conn.cursor()

    # 1. Localiza a obrigaÃ§Ã£o real do IPVA
    cur.execute("""
        SELECT id, data, descricao, valor, pago
        FROM transacoes
        WHERE (descricao LIKE '%IPVA%' OR categoria LIKE '%IPVA%')
          AND tipo = 'despesa'
        ORDER BY id DESC LIMIT 1
    """)
    row_ipva = cur.fetchone()

    if not row_ipva:
        return {"tem_missao": False, "mensagem": "Nenhuma obrigaÃ§Ã£o de IPVA encontrada no banco."}

    t_id, t_data, t_desc, t_valor, t_pago = row_ipva
    dt_venc_padrao = datetime.strptime(t_data, "%Y-%m-%d").date() if isinstance(t_data, str) else t_data
    valor_nominal_padrao = to_decimal(t_valor)

    # 2. Resumo de pagamentos reais jÃ¡ efetuados no banco
    resumo_obrig = saldo_service.obter_resumo_obrigacao(conn, t_id)
    valor_ja_pago = resumo_obrig["total_pago"]

    # 3. Busca meta e reserva virtual cadastrada em planejamento_metas
    cur.execute("""
        SELECT id, valor_reservado, turnos_planejados, horas_por_turno,
               ganho_bruto_hora_estimado, combustivel_hora_estimado,
               status, prioridade, valor_alvo, data_limite, titulo
        FROM planejamento_metas
        WHERE transacao_alvo_id = ? AND mes_ano = ?
    """, (t_id, mes_ano))
    row_meta = cur.fetchone()

    if row_meta:
        meta_id = row_meta[0]
        valor_reservado = to_decimal(row_meta[1])
        turnos_planejados = int(row_meta[2] or 5)
        horas_turno = float(row_meta[3] or 4.0)
        ganho_hora = float(row_meta[4] or 35.0)
        comb_hora = float(row_meta[5] or 10.0)
        status = str(row_meta[6] or 'ativa')
        prioridade = str(row_meta[7] or 'alta')
        valor_alvo = to_decimal(row_meta[8]) if row_meta[8] else valor_nominal_padrao
        data_limite_str = str(row_meta[9]) if row_meta[9] else str(t_data)
        titulo = str(row_meta[10]) if row_meta[10] else str(t_desc)
    else:
        meta_id = None
        valor_reservado = Decimal("0.00")
        turnos_planejados = 5
        horas_turno = 4.0
        ganho_hora = 35.0
        comb_hora = 10.0
        status = 'ativa'
        prioridade = 'alta'
        valor_alvo = valor_nominal_padrao
        data_limite_str = str(t_data)
        titulo = str(t_desc)

    dt_venc = datetime.strptime(data_limite_str, "%Y-%m-%d").date() if isinstance(data_limite_str, str) else data_limite_str

    # 3.1 Cofrinho IPVA (Fundo IPVA) - alimenta automaticamente a reserva sem lanÃ§ar duas vezes
    cur.execute("SELECT name FROM sqlite_master WHERE type='table' AND name='cofrinhos'")
    if cur.fetchone() is not None:
        cur.execute("""
            SELECT id, saldo_confirmado, conta_config_id 
            FROM cofrinhos 
            WHERE nome LIKE '%IPVA%' 
            ORDER BY id ASC LIMIT 1
        """)
        row_cof_ipva = cur.fetchone()
        saldo_cof_ipva = Decimal("0.00")
        if row_cof_ipva:
            if row_cof_ipva[2]:
                try:
                    res_sc = saldo_service.calcular_saldo_conta(conn, row_cof_ipva[2])
                    saldo_cof_ipva = res_sc["saldo_atual"]
                except Exception:
                    saldo_cof_ipva = to_decimal(row_cof_ipva[1])
            else:
                saldo_cof_ipva = to_decimal(row_cof_ipva[1])
        valor_reservado = max(valor_reservado, saldo_cof_ipva)

    # 4. CÃ¡lculos de Saldo e Prazo
    # O que falta cobrir: valor alvo menos o que jÃ¡ foi pago e menos o que jÃ¡ estÃ¡ reservado
    saldo_devedor_restante = max(Decimal("0.00"), valor_alvo - valor_ja_pago - valor_reservado)
    dias_restantes = max(0, (dt_venc - data_referencia).days)

    meta_por_turno = (
        saldo_devedor_restante / Decimal(str(max(1, turnos_planejados)))
    ).quantize(CENTAVOS, rounding=ROUND_HALF_EVEN)

    percentual_coberto = (
        ((valor_ja_pago + valor_reservado) / valor_alvo * 100)
        if valor_alvo > 0 else Decimal("100.00")
    )

    return {
        "tem_missao": True,
        "meta_id": meta_id,
        "transacao_id": t_id,
        "titulo": titulo,
        "data_vencimento": data_limite_str,
        "dias_restantes": dias_restantes,
        "valor_nominal": valor_alvo,
        "valor_ja_pago": valor_ja_pago,
        "valor_reservado": valor_reservado,
        "saldo_devedor_restante": saldo_devedor_restante,
        "percentual_coberto": float(percentual_coberto),
        "turnos_planejados": turnos_planejados,
        "horas_por_turno": horas_turno,
        "meta_por_turno": meta_por_turno,
        "ganho_hora_estimado": ganho_hora,
        "comb_hora_estimado": comb_hora,
        "status": status,
        "prioridade": prioridade,
        "is_ativa": status == "ativa",
        "is_concluida": status == "concluida"
    }


def atualizar_missao_parametros(
    conn: sqlite3.Connection,
    transacao_id: int,
    valor_alvo: Decimal,
    data_limite: str,
    prioridade: str = "alta",
    valor_reserva: Decimal = Decimal("0.00"),
    turnos_planejados: int = 5,
    horas_por_turno: float = 4.0,
    status: str = "ativa",
    titulo: Optional[str] = None,
    mes_ano: Optional[str] = None
) -> Tuple[bool, str]:
    """
    Edita os parÃ¢metros da missÃ£o: valor, vencimento, prioridade, reserva virtual, turnos e status.
    Garante consistÃªncia: nÃ£o cria pagamentos fictÃ­cios e valida se a reserva virtual nÃ£o ultrapassa o saldo disponÃ­vel.
    """
    if mes_ano is None:
        mes_ano = date.today().strftime("%Y-%m")

    saldo_info = saldo_service.calcular_saldo_consolidado(conn)
    saldo_disp = saldo_info["saldo_consolidado"]

    if valor_reserva < Decimal("0.00"):
        return False, "O valor da reserva nÃ£o pode ser negativo."

    if valor_reserva > saldo_disp:
        return False, f"Reserva (R$ {valor_reserva:.2f}) excede o saldo disponÃ­vel total em caixa (R$ {saldo_disp:.2f})."

    cur = conn.cursor()
    cur.execute("""
        SELECT id FROM planejamento_metas
        WHERE transacao_alvo_id = ? AND mes_ano = ?
    """, (transacao_id, mes_ano))
    row = cur.fetchone()

    val_alvo_flt = float(valor_alvo)
    val_res_flt = float(valor_reserva)

    if row:
        meta_id = row[0]
        cur.execute("""
            UPDATE planejamento_metas
            SET valor_alvo = ?,
                data_limite = ?,
                prioridade = ?,
                valor_reservado = ?,
                turnos_planejados = ?,
                horas_por_turno = ?,
                status = ?,
                titulo = COALESCE(?, titulo)
            WHERE id = ?
        """, (val_alvo_flt, str(data_limite), prioridade, val_res_flt, turnos_planejados, horas_por_turno, status, titulo, meta_id))
    else:
        if not titulo:
            cur.execute("SELECT descricao FROM transacoes WHERE id = ?", (transacao_id,))
            t_row = cur.fetchone()
            titulo = t_row[0] if t_row else "MissÃ£o IPVA"

        cur.execute("""
            INSERT INTO planejamento_metas (
                mes_ano, titulo, transacao_alvo_id, valor_alvo, valor_reservado,
                data_limite, turnos_planejados, horas_por_turno, status, prioridade
            ) VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?)
        """, (mes_ano, titulo, transacao_id, val_alvo_flt, val_res_flt, str(data_limite), turnos_planejados, horas_por_turno, status, prioridade))

    conn.commit()
    return True, f"MissÃ£o '{titulo}' atualizada com sucesso!"


def atualizar_status_missao(
    conn: sqlite3.Connection,
    transacao_id: int,
    novo_status: str,
    mes_ano: Optional[str] = None
) -> Tuple[bool, str]:
    """
    Atualiza o estado de uma missÃ£o ('ativa', 'pausada', 'concluida', 'arquivada').
    Quando concluÃ­da (quitada), a missÃ£o deixa de aparecer como pendÃªncia ativa, permanecendo no histÃ³rico.
    NÃ£o cria pagamentos fictÃ­cios nem altera a transaÃ§Ã£o bancÃ¡ria original.
    """
    if novo_status not in ('ativa', 'pausada', 'concluida', 'arquivada'):
        return False, f"Status invÃ¡lido: '{novo_status}'."

    if mes_ano is None:
        mes_ano = date.today().strftime("%Y-%m")

    cur = conn.cursor()
    cur.execute("""
        SELECT id FROM planejamento_metas
        WHERE transacao_alvo_id = ? AND mes_ano = ?
    """, (transacao_id, mes_ano))
    row = cur.fetchone()

    if row:
        cur.execute("UPDATE planejamento_metas SET status = ? WHERE id = ?", (novo_status, row[0]))
    else:
        cur.execute("SELECT data, descricao, valor FROM transacoes WHERE id = ?", (transacao_id,))
        t_row = cur.fetchone()
        dt_venc = t_row[0] if t_row else "2026-10-29"
        tit = t_row[1] if t_row else "MissÃ£o IPVA"
        val_nom = float(t_row[2]) if t_row else 431.11

        cur.execute("""
            INSERT INTO planejamento_metas (mes_ano, titulo, transacao_alvo_id, valor_alvo, valor_reservado, data_limite, status)
            VALUES (?, ?, ?, ?, 0.0, ?, ?)
        """, (mes_ano, tit, transacao_id, val_nom, dt_venc, novo_status))

    conn.commit()
    return True, f"Status da missÃ£o alterado para '{novo_status}' com sucesso!"

# COFRINHOS REAIS (MERCADO PAGO / ENVELOPES)
# =============================================================================
def obter_cofrinhos(conn: sqlite3.Connection) -> List[Dict[str, Any]]:
    """
    Retorna a lista de cofrinhos reais cadastrados no Mercado Pago.
    Calcula dinamicamente o saldo com base nas transferÃªncias internas para a subconta vinculada.
    """
    cur = conn.cursor()
    cur.execute("PRAGMA table_info(cofrinhos)")
    cols = [r[1] for r in cur.fetchall()]
    tem_cc = "conta_config_id" in cols
    tem_inc = "incluir_saldo_consolidado" in cols

    query = "SELECT id, conta_pai_id, nome, saldo_confirmado, meta_valor, objetivo, data_atualizacao, observacoes"
    if tem_cc:
        query += ", conta_config_id"
    else:
        query += ", NULL as conta_config_id"
    if tem_inc:
        query += ", incluir_saldo_consolidado"
    else:
        query += ", 0 as incluir_saldo_consolidado"
    query += " FROM cofrinhos ORDER BY id ASC"

    cur.execute(query)
    rows = cur.fetchall()
    res = []
    for r in rows:
        cof_id = r[0]
        c_pai_id = r[1]
        c_nome = r[2]
        s_conf = to_decimal(r[3])
        m_val = to_decimal(r[4])
        obj = r[5] or ""
        dt_at = r[6] or ""
        obs = r[7] or ""
        cc_id = r[8]
        inc_cons = r[9] if len(r) > 9 and r[9] is not None else 0

        # Se houver conta_config vinculada, calcula saldo contÃ¡bil exato
        saldo_real = s_conf
        if cc_id:
            try:
                calc = saldo_service.calcular_saldo_conta(conn, cc_id)
                saldo_real = calc["saldo_atual"]
            except Exception:
                saldo_real = s_conf

        res.append({
            "id": cof_id,
            "conta_pai_id": c_pai_id,
            "nome": c_nome,
            "saldo_confirmado": saldo_real,
            "meta_valor": m_val,
            "objetivo": obj,
            "data_atualizacao": dt_at,
            "observacoes": obs,
            "conta_config_id": cc_id,
            "incluir_saldo_consolidado": inc_cons
        })
    return res



def atualizar_saldo_cofrinho(
    conn: sqlite3.Connection,
    cofrinho_id: int,
    novo_saldo: Decimal,
    observacoes: Optional[str] = None
) -> Tuple[bool, str]:
    """
    Atualiza o saldo confirmado de um cofrinho real do Mercado Pago informado manualmente pelo usuÃ¡rio.
    NÃ£o duplica com o saldo geral da conta Mercado Pago.
    """
    if novo_saldo < Decimal("0.00"):
        return False, "O saldo do cofrinho nÃ£o pode ser negativo."

    cur = conn.cursor()
    dt_hoje = date.today().strftime("%Y-%m-%d")
    val_flt = float(novo_saldo)
    cur.execute("""
        UPDATE cofrinhos
        SET saldo_confirmado = ?,
            data_atualizacao = ?,
            observacoes = COALESCE(?, observacoes)
        WHERE id = ?
    """, (val_flt, dt_hoje, observacoes, cofrinho_id))
    conn.commit()
    return True, f"Saldo do cofrinho atualizado para R$ {novo_saldo:.2f} com sucesso!"


def criar_ou_editar_cofrinho(
    conn: sqlite3.Connection,
    nome: str,
    meta_valor: Decimal,
    objetivo: str,
    cofrinho_id: Optional[int] = None
) -> Tuple[bool, str]:
    """
    Permite cadastrar ou editar cofrinhos reais manualmente.
    """
    nome_limpo = nome.strip()
    if not nome_limpo:
        return False, "O nome do cofrinho nÃ£o pode ser vazio."

    cur = conn.cursor()
    meta_flt = float(meta_valor)
    dt_hoje = date.today().strftime("%Y-%m-%d")

    if cofrinho_id:
        cur.execute("""
            UPDATE cofrinhos
            SET nome = ?, meta_valor = ?, objetivo = ?
            WHERE id = ?
        """, (nome_limpo, meta_flt, objetivo.strip(), cofrinho_id))
        conn.commit()
        return True, f"Cofrinho '{nome_limpo}' atualizado com sucesso!"
    else:
        try:
            cur.execute("""
                INSERT INTO cofrinhos (conta_pai_id, nome, saldo_confirmado, meta_valor, objetivo, data_atualizacao)
                VALUES (1, ?, 0.0, ?, ?, ?)
            """, (nome_limpo, meta_flt, objetivo.strip(), dt_hoje))
            conn.commit()
            return True, f"Cofrinho '{nome_limpo}' cadastrado com sucesso!"
        except sqlite3.IntegrityError:
            return False, f"JÃ¡ existe um cofrinho com o nome '{nome_limpo}'."

# 5. SIMULAÃ‡ÃƒO DE METAS DE TRABALHO DO MOTORISTA & VIABILIDADE RIGOROSA
# =============================================================================
def calcular_viabilidade_metas_trabalho(
    necessidade_financeira: Decimal,
    dias_disponiveis: int = 5,
    horas_disponiveis_dia: float = 4.0,
    ganho_bruto_hora: float = 35.0,
    combustivel_hora: float = 10.0
) -> Dict[str, Any]:
    """
    Calcula a viabilidade operacional do motorista para cobrir a necessidade financeira.
    Regra de rigor exigida pelo usuÃ¡rio:
    Se preciso de 21 horas, mas tenho apenas 20 disponÃ­veis, a meta NÃƒO estÃ¡ coberta (dÃ©ficit de 1 hora).
    """
    ganho_liq_hora = max(1.0, ganho_bruto_hora - combustivel_hora)
    liq_decimal = to_decimal(ganho_liq_hora)

    horas_disponiveis_total = float(dias_disponiveis * horas_disponiveis_dia)
    
    # Horas necessÃ¡rias para cobrir o valor em dinheiro
    if necessidade_financeira <= Decimal("0.00"):
        horas_necessarias = 0.0
    else:
        horas_necessarias = float(necessidade_financeira / liq_decimal)

    diferenca_horas = horas_disponiveis_total - horas_necessarias

    # ClassificaÃ§Ã£o rigorosa de viabilidade com linguagem prudente e estimativa
    if horas_necessarias <= 0:
        status_viabilidade = "coberta_folga"
        rotulo_viabilidade = "Estimativa: Sem necessidade adicional de caixa para compromissos mÃ­nimos confirmados"
        cor_badge = "green"
    elif horas_necessarias <= horas_disponiveis_total:
        folga = diferenca_horas
        status_viabilidade = "coberta"
        rotulo_viabilidade = f"Estimativa Preliminar: Meta coberta com margem estimada de {folga:.1f}h"
        cor_badge = "green"
    else:
        deficit = abs(diferenca_horas)
        status_viabilidade = "deficit"
        rotulo_viabilidade = f"Estimativa Preliminar: ProjeÃ§Ã£o de dÃ©ficit estimado de {deficit:.1f}h de volante"
        cor_badge = "red"

    return {
        "necessidade_financeira": necessidade_financeira,
        "ganho_bruto_hora": ganho_bruto_hora,
        "combustivel_hora": combustivel_hora,
        "ganho_liquido_hora": ganho_liq_hora,
        "dias_disponiveis": dias_disponiveis,
        "horas_disponiveis_dia": horas_disponiveis_dia,
        "horas_disponiveis_total": horas_disponiveis_total,
        "horas_necessarias": round(horas_necessarias, 1),
        "diferenca_horas": round(diferenca_horas, 1),
        "status_viabilidade": status_viabilidade,
        "rotulo_viabilidade": rotulo_viabilidade,
        "cor_badge": cor_badge,
        "modo_estimativa": True,
        "aviso_metodologico": "ProjeÃ§Ã£o preliminar baseada em estimativas operacionais. CondiÃ§Ãµes reais de trÃ¢nsito e demanda podem variar."
    }
